\section{Úvod}
\begin{frame}[fragile=singleslide]\frametitle{Digitalizácia hudby}
\begin{itemize}
	\item Prístup k hudbe sa vďaka internetu rozšíril na celý svet
	\item Umožnili tomu hlavne streamovacie platformy ako napríklad Spotify, Youtube Music a Apple Music
	\item Umelci teraz môžu osloviť publikum na celom svete, čím sa otvárajú nové príležitosti na trhu hudby
\end{itemize}
\end{frame}

\section{Globalizácia hudby}
\begin{frame}[fragile=singleslide]\frametitle{Dosah hudby a jej vplyv}
\begin{itemize}
	\item Umelci svoju hudbu môžu zdieľať s kýmkoľvek pripojeným na internet
	\item Objavili sa nové trhy s hudbou, vďaka ktorým sa menším umelcom ľahšie darí.
	\item Táto zmena urobila z hudby skutočne globálny priemysel
\end{itemize}
\end{frame}

\section{Odporúčací systém}
\begin{frame}[fragile=singleslide]\frametitle{Ako sa vieme k hudbe dostať?}
\begin{itemize}
	\item 
	\item S neobmedzeným prístupom k hudbe nastáva problém výberu, ktorý rieši práve recommendation systém.
	\item Na základe histórie hudby, ktorú sme počúvali, vie pomocou rôznych algoritmov odporúčiť nové skladby.
	\item Vďaka tomuto systému je výber hudby osobnejší, pretože sa odvíja od našich individuálnych preferencií.
\end{itemize}
\end{frame}

\begin{frame}[fragile=singleslide]\frametitle{Ako odporúčací system funguje?}
\begin{itemize}
	\item Collaborative filtering
	\item Content-based filtering
	\item Natural language models
	\item Temporal analysis
	\item User data analysis
\end{itemize}
\end{frame}

\section{Určovanie trendov}
\begin{frame}[fragile=singleslide]\frametitle{Dôsledky pre umelcov}
\begin{itemize}
	\item Umelci pre úspech musia hudbu upravovať aby vyhovovala odporúčaciemu systému
	\item Toto vedia k tvorbe trendov, ktoré vlastne závisia od odporúčacieho systému
	\item Týmto sa obmedzuje voľnosť kreativity umelcov alternatívnych žánrov
\end{itemize}
\end{frame}

\begin{frame}\frametitle{Dôsledky pre poslucháčov}
\begin{itemize}
	\item Systém ma problém odporučiť novú hudbu pre nepopulárne žánre.
	\item Cyklické odporúčanie, kvôli zlému vyhodnoteniu dát
	\item Zužovanie žánrov a krabičkovanie hudby
\end{itemize}
\end{frame}
